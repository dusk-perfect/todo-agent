## 一、做了什么

在小明只会回「你好，欢迎使用小明智能体」的模板上，改造成能用中文管待办的助手。整体分三层：

| 层 | 位置 | 职责 |
| --- | --- | --- |
| 规则层 | `server/agent/intent.ts` 等 | 中文意图识别，不配置任何 Key 也能完成全部验收动作 |
| 模型层 | `server/agent/llm.ts` | 规则接不住的句子才交给 DeepSeek，校验不过或超时就退回规则 |
| 存取层 | `server/task-store.ts` `server/users.ts` 等 | 任务落在本机 JSON，原子写；按匿名身份分目录 |

意图识别的顺序是**规则优先、模型兜底**：规则判得快且确定，模型只在规则确实听不懂时才上场，没有 Key 时全程走规则。

```ts
import {

import { planWithLLM } from "./agent/llm.ts";

function listBlock(tasks: TaskStore): string {
  return formatResults([{ action: { type: "list" }, tasks: tasks.list() }], tasks.list());
}

// 把用户明确要补充的那句写进上一条任务：属性拆出来，剩下的记成 detail。
function writeDetail(text: string, task: Task, tasks: TaskStore): string {
  const fields: Record<string, unknown> = {};
  // 「明天内完成」这种截止说法整段算日期，别留在备注里
  const stripped = stripDeadlineSegments(text.trim());
  if (stripped.dueAt) fields.dueAt = stripped.dueAt;
  let remainder = stripped.rest;

  const priority = remainder.match(/(高|中|低)优先级/);
  if (priority) {
    fields.priority = { 高: "high", 中: "medium", 低: "low" }[priority[1] as "高" | "中" | "低"];
    remainder = remainder.replace(priority[0], "");
  }

  const estimate = remainder.match(/(大|中|小)工作量/);
  if (estimate) {
    fields.estimate = { 大: "large", 中: "medium", 小: "small" }[estimate[1] as "大" | "中" | "小"];
    remainder = remainder.replace(estimate[0], "");
  }

  const due = parseDueDate(remainder);
  if (due) {
    fields.dueAt ??= due.date;
    remainder = remainder.replace(due.matched, "");
  }

  // 补的时刻也记进 dueTime，备注原文不动，免得用户看不到自己说的时间
  const clock = parseDueTime(remainder);
  if (clock) fields.dueTime = clock.time;

  remainder = remainder.replace(/(?:到期|截止)/g, "").replace(/^[，,、\s]+|[，,、\s]+$/g, "");
  if (remainder) {
    // 已经补过的细节保留，新的接在后面，不覆盖
    const existing = typeof task.detail === "string" ? task.detail : "";
    fields.detail =
      existing && !existing.includes(remainder) ? `${existing}；${remainder}` : remainder;
  }

  if (Object.keys(fields).length === 0) {
    return `「${task.title}」先按现在的记录保存。\n\n${listBlock(tasks)}`;
  }

  executeActions(tasks, [{ type: "setAttrs", id: task.id, fields }]);
  const summary =
    typeof fields.detail === "string"
      ? `已为「${task.title}」记下备注：${fields.detail}`
      : `已更新「${task.title}」的属性`;
  return `${summary}\n\n${listBlock(tasks)}`;
}

// 接收本轮对话历史与任务存储，返回小明的回复。
// 意图识别顺序：规则 → 模型 → 兜底；规则优先是为了确定和快，模型只在规则听不懂时兜底。
export async function reply(messages: ChatMessage[], tasks: TaskStore): Promise<string> {
  const latest = messages.at(-1);
  if (!latest || latest.role !== "user") return "请告诉我你想做什么。";

  const snapshot = tasks.list();
  const ruleActions = parseRuleActions(latest.content, snapshot);

  // 上一轮刚追问过细节：只有用户明确说要补充，才写进上一条任务。
  // 说「没有」就结束；没说要补充（比如「我现在要去取快递」）一律按另一件事处理。
  const pending = pendingDetailTask(messages, snapshot);
  // 追问期间说了别的指令（完成、删除、另一件事），要说明上一条的备注先留空
  let droppedDetailTitle: string | null = null;
  if (pending) {
    const answer = parseDetailAnswer(latest.content);
    if (answer.kind === "none") {
      return `好的，「${pending.title}」先按现在的记录保存。\n\n${listBlock(tasks)}`;
    }
    if (answer.kind === "agree") return writeDetail(answer.text, pending, tasks);
    if (answer.kind === "ask") return detailContentPrompt(pending.title);
    droppedDetailTitle = pending.title;
  }

  // 上一轮刚问过「哪天完成」：这一句先说日期，就补到那条任务上
  const duePending = pendingDueTask(messages, snapshot);
  if (duePending) {
    const answer = parseDueAnswer(latest.content);
    const date =
      answer.kind === "date" ? answer.date : answer.kind === "today" ? daysFromToday(0) : null;
    if (date) {
      const fields: Record<string, unknown> = { dueAt: date };
      if (answer.kind !== "other" && answer.time) fields.dueTime = answer.time;
      executeActions(tasks, [{ type: "setAttrs", id: duePending.id, fields }]);
      const time = typeof fields.dueTime === "string" ? ` ${fields.dueTime}` : "";
      return `「${duePending.title}」记到 ${dueBadge(date)}${time}。\n\n${listBlock(tasks)}`;
    }
    // 说的是别的事：这条先按当天算，再照常处理这句话
    executeActions(tasks, [
      { type: "setAttrs", id: duePending.id, fields: { dueAt: daysFromToday(0) } },
    ]);
  }

  let actions: Action[] | null = ruleActions;
  if (!actions) actions = await planWithLLM(latest.content, snapshot);

  // 没有待追问的时候回一句「没有」「算了」，就是没有别的要记了，
  // 不该被当成一件事，也不该回一句「我可以帮你……」。
  if (!actions && isNoDetailAnswer(latest.content)) {
    return `好的，没有别的要记的了。\n\n${listBlock(tasks)}`;
  }

  // 只说了一个时间、没有上下文（「就是今天啊」「等下肯定也是今天」）：
  // 当成一句确认，别当成新任务，也别再兜底。
  if (!actions) {
    const asserted = parseDueAnswer(latest.content);
    if (asserted.kind === "today" || asserted.kind === "date") {
      const label = asserted.kind === "today" ? "今天" : dayLabel(asserted.date);
      return `好的，那就按${label}算。\n\n${listBlock(tasks)}`;
    }
  }

  // 都没有结果时：长句压成一条整体建出来，别让用户白说一场；
  // 追问状态下的任何非问句都按「没说要补充，就算另一件事」处理。问题类句子不建任务。
  if (!actions && !isQuestion(latest.content)) {
    const title =
      parseLongStatementTitle(latest.content) ??
      (pending ? latest.content.trim().slice(0, 200) : null);
    if (title) {
      // 兜底建档也要把日期、优先级、工作量解出来，别整句塞进标题
      const { rest, attrs } = extractAttributes(title);
      actions = [{ type: "create", title: rest || title, fields: attrs }];
    }
  }
  if (!actions) actions = [{ type: "ask", question: FALLBACK_QUESTION }];

  // 没提日期默认当天；但刚说过别的日期、这句又没说，就先问清楚，别默认错。
  // 「早上/中午/下午」这类只说时段的说法本身就限定在当天，等价于说了今天。
  const createCount = actions.filter((action) => action.type === "create").length;
  const today = daysFromToday(0);
  const statedDate = statedDay(latest.content);
  const inherited = lastMentionedDate(messages);
  let askDueTitle: string | null = null;

  actions = actions.map((action) => {
    if (action.type !== "create") return action;
    const fields = action.fields ?? {};
    if (typeof fields.dueAt === "string") return action;
    // 这句话里说过的日期对句中每条任务都算数：
    // 「下周三之前把复盘写完，然后发给张老师」两条都该落在下周三
    if (statedDate) return { ...action, fields: { ...fields, dueAt: statedDate } };
    if (createCount === 1 && inherited && inherited !== today) {
      askDueTitle = action.title;
      return action;
    }
    return { ...action, fields: { ...fields, dueAt: today } };
  });

  const results = executeActions(tasks, actions);
  const created = results.filter((result) => result.action.type === "create" && result.task);
  // 新任务如果落在有日程的日子上，顺手提醒一句，免得排重了
  const busyNote = created
    .map((result) =>
      typeof result.task?.dueAt === "string" ? calendarNote(result.task.dueAt) : "",
    )
    .filter(Boolean)
    .join("");
  const answer = formatResults(results, tasks.list()) + busyNote;
  const asked = results.some((result) => result.action.type === "ask");

  if (created.length === 1) {
    const title = created[0].task?.title;
    // 日期还没定：先问哪天，别紧接着又追问备注，一次只问一件事
    if (title && askDueTitle) return answer + duePrompt(title);
    // 已经有一个反问在等用户回答，别再叠一个追问
    if (title && asked) return answer;
    // 追问被打断时说明一句，免得用户以为刚才那条被改了
    if (title && droppedDetailTitle) {
      return (
        `这算另一件事，「${droppedDetailTitle}」的备注先留空。\n\n${answer}` + detailPrompt(title)
      );
    }
    if (title) return answer + detailPrompt(title);
  }
  // 本轮做的是完成、删除这类操作，但上一轮在等备注：说明备注先留空。
  // 只是问一句清单不算打断，不提上一条，免得答非所问。
  const mutating = actions.some((action) =>
    ["create", "complete", "reopen", "rename", "setAttrs", "delete"].includes(action.type),
  );
  if (droppedDetailTitle && mutating) {
    return `「${droppedDetailTitle}」的备注先留空。\n\n${answer}`;
  }
  return answer;
}

const QUESTION_RE = /[？?]\s*$|^(?:为什么|怎么|多少|几点|什么|谁|哪里|哪个|是不是|能不能)/;

// 最近几轮用户提过的日期。用来判断「接着说的这条」是不是该问清楚哪天：
// 上一条说的是明天，下一条不吭声就记成今天，多半是记错了。
// 只说了时段（「下午」）的那条就是当天，同样算一次明确的日期交代。
function lastMentionedDate(messages: ChatMessage[]): string | null {
  const history = messages.slice(0, -1);
  let seen = 0;
  for (let index = history.length - 1; index >= 0; index -= 1) {
    const message = history[index];
    if (!message || message.role !== "user") continue;
    seen += 1;
    if (seen > 4) break;
    const day = statedDay(message.content);
    if (day) return day;
  }
  return null;
}

function isQuestion(text: string): boolean {
  return QUESTION_RE.test(text.trim());
}
```

```ts
import type { Task, TaskStore } from "../task-store.ts";

export type Action =
  | { type: "create"; title: string; fields?: Record<string, unknown> }
  | { type: "list" }
  | { type: "due"; date: string }
  | { type: "complete"; id: string }
  | { type: "reopen"; id: string }
  | { type: "rename"; id: string; title: string }
  | { type: "setAttrs"; id: string; fields: Record<string, unknown> }
  | { type: "delete"; id: string }
  | { type: "ask"; question: string };

export interface ActionResult {
  action: Action;
  task?: Task;
  tasks?: Task[];
}

export function executeActions(tasks: TaskStore, actions: Action[]): ActionResult[] {
  return actions.map((action) => {
    switch (action.type) {
      case "create":
        return { action, task: tasks.create(action.title, action.fields) };
      case "list":
        return { action, tasks: tasks.list() };
      case "due":
        return {
          action,
          tasks: tasks.list().filter((task) => task.dueAt === action.date && !task.done),
        };
      case "setAttrs":
        return { action, task: tasks.update(action.id, action.fields) };
      case "complete":
        return { action, task: tasks.update(action.id, { done: true }) };
      case "reopen":
        return { action, task: tasks.update(action.id, { done: false }) };
      case "rename":
        return { action, task: tasks.update(action.id, { title: action.title }) };
      case "delete": {
        const task = tasks.list().find((item) => item.id === action.id);
        tasks.remove(action.id);
        return { action, task };
      }
      case "ask":
        return { action };
    }
  });
}
```

```ts
// 而是剥掉「主语 + 时间 + 意愿词 + 去」这些封闭的前缀，剩下的就是事情本身。

  /(.+?)(?:改到|改成|设为|设成|定在|安排到|挪到)(今天|今日|明天|明日|后天|大后天|下{1,2}周[一二三四五六日天]|\d{1,2}月\d{1,2}[日号]?)(?:之内|以内|内)?(?:完成|做完|搞定)?$/,

const PRIORITY = { 高: "high", 中: "medium", 低: "low" } as const;
const ESTIMATE = { 大: "large", 中: "medium", 小: "small" } as const;

function normalize(text: string): string {
  return text
    .replace(/[\s，。！？、；：,.!?;:]/g, "")
    .replace(/(这条|那个|这个|一件|一下|这个任务|这条任务)/g, "");
}

function longestCommonSubstring(a: string, b: string): string {
  let best = "";
  let previous = Array.from({ length: b.length + 1 }, () => 0);
  for (let i = 1; i <= a.length; i += 1) {
    const current = [0];
    for (let j = 1; j <= b.length; j += 1) {
      if (a[i - 1] === b[j - 1]) {
        current[j] = previous[j - 1] + 1;
        if (current[j] > best.length) best = a.slice(i - current[j], i);
      } else {
        current[j] = 0;
      }
    }
    previous = current;
  }
  return best;
}

function referenceScore(message: string, title: string): number {
  const m = normalize(message);
  const t = normalize(title);
  if (!m || !t) return 0;
  if (m === t) return 1;
  if (m.includes(t)) return 0.95;
  // 用户说的是任务名的一部分（「报告」指「下午交报告」）：也算高度匹配，
  // 但比完整相等低一档，两条都像的时候好反问而不是替用户挑
  if (t.includes(m)) return 0.93;
  const common = longestCommonSubstring(m, t);
  if (common.length < 2) return 0;
  return 0.5 + (0.4 * common.length) / Math.min(m.length, t.length);
}

interface TaskMatch {
  task?: Task;
  candidates: Task[];
  ambiguous: boolean;
}

function findTask(message: string, tasks: Task[]): TaskMatch {
  const ranked = tasks
    .map((task) => ({ task, score: referenceScore(message, task.title) }))
    .filter((entry) => entry.score >= 0.5)
    .sort((a, b) => b.score - a.score);

  if (ranked.length === 0) return { candidates: [], ambiguous: false };
  if (ranked.length > 1 && ranked[0].score - ranked[1].score < 0.05) {
    return { candidates: ranked.slice(0, 3).map((entry) => entry.task), ambiguous: true };
  }
  return { task: ranked[0].task, candidates: [ranked[0].task], ambiguous: false };
}

// 标题开头的废话：「我」「等下」「要」这些说的是谁、什么时候、什么语气，
// 不是事情本身，剥掉后剩下的才是要做的事。
const LEADING_NOISE_RE =
  /^(?:一下|一条|一个|一件|把|吧|呢|啊|哦|哈|呀|我|咱|俺|我们|现在|马上|立刻|等下|待会|待会儿|一会儿|晚点|稍后|要|得|需要)+/;

function cleanTitle(raw: string): string {
  let title = raw.trim();
  for (;;) {
    const before = title;
    title = title.replace(LEADING_NOISE_RE, "").trim();
    if (title === before) break;
  }
  return title
    .replace(/(?:吧|呢|啊|哦|哈|呀)+$/, "")
    .replace(/^(?:的|了)+/, "")
    .trim();
}

// 「删除待办」「清空任务」里没有具体标题，按指代处理，别当成一个名字叫「待办」的任务
const GENERIC_REFERENCE_RE = /^(?:待办|任务|清单|列表|东西|事项|全部|所有)$/;

// 「把买牛奶这条待办删掉」里的「这条待办」是说话习惯，不是标题的一部分。
// 只在直接匹配失败时兜底再试一次，避免把真叫「交任务」的任务削成「交」。
function stripTaskSuffix(text: string): string {
  return text
    .replace(/(?:这|那|该)?(?:一)?(?:条|个|件|项)?(?:待办|任务|事项?)$/, "")
    .replace(/(?:这|那|该)(?:一)?(?:条|个|件|项)$/, "")
    .replace(/的$/, "")
    .trim();
}

// 只有时间的片段：「下午三点半」「明天」「周五」，本身不是一件事
function isTimeOnly(text: string): boolean {
  let left = text.trim();
  for (;;) {
    const before = left;
    left = left.replace(DATE_PREFIX_RE, "").replace(CLOCK_PREFIX_RE, "");
    if (left === before) break;
  }
  return left.replace(/[，,、：:\s]/g, "") === "";
}

// 陈述句里的「和」多半是介词（「和小东开会」），不是并列，硬拆会把一句话劈成半句
// 左边以助动词收尾，说明「和」是介词不是并列：「我要和朋友出去玩」是一件事
const AUX_END_RE = /(?:要|得|需要|想|会|能|可以|应该|必须|帮|给)$/;

function splitTitles(raw: string, allowAnd = true): string[] {
  const text = raw.trim();
  if (!text) return [];

  // 「下午三点半和小东开会」：切开一侧只剩时间，说明「和」后面才是事情，别拆
  if (allowAnd) {
    const aroundAnd = text.split(/\s*和\s*/);
    if (aroundAnd.length > 1 && aroundAnd.some(isTimeOnly)) {
      const title = cleanTitle(text);
      return title ? [title] : [];
    }
  }

  // 长句不按标点硬拆：拆错的代价（好几条垃圾任务）比不拆大。
  // 短句才按「和/、/再/然后」拆成多条。
  if (text.length > 16) {
    const head = text.split(/[，,；;。]/)[0] ?? text;
    const title = cleanTitle(head);
    return title ? [title] : [];
  }

  // 「和」只有在两边都说得完整时才是并列，否则按介词处理
  const andIndex = text.indexOf("和");
  const andIsConjunction =
    allowAnd && andIndex > 0 && !AUX_END_RE.test(text.slice(0, andIndex).trim());
  const separator = andIsConjunction
    ? /\s*(?:和|以及|、|,|，|;|；|然后|还有|顺便)\s*/
    : /\s*(?:以及|、|,|，|;|；|然后|还有|顺便)\s*/;
  return text
    .split(separator)
    .map(cleanTitle)
    .filter((title) => title.length > 0);
}

// 「我现在要去取快递」→ 去取快递；「这周末要给车做保养」→ 给车做保养。
// 必须削掉过意愿词（要/得/需要/准备…）才算陈述式任务，否则「今天天气怎么样」也会被当成任务。
// 时刻（下午三点、上午十点）不能削：dueAt 只到天，削掉就真丢了，先摘出来最后拼回标题。
function statementTitle(message: string): string | null {
  let rest = message.trim();
  const subject = rest.match(SUBJECT_PREFIX_RE);
  if (subject) rest = rest.slice(subject[0].length);

  let sawModal = false;
  let clock = "";
  // 句首交代了哪一天（今天/明天/下周三…）：带日期的陈述本身就是一张待办
  let sawDate = false;
  for (;;) {
    const before = rest;
    const clockMatch = rest.match(CLOCK_PREFIX_RE);
    if (clockMatch) {
      clock += clockMatch[0];
      rest = rest.slice(clockMatch[0].length);
    }
    if (DATE_PREFIX_RE.test(rest)) sawDate = true;
    rest = rest.replace(DATE_PREFIX_RE, "");
    rest = rest.replace(INSTANT_PREFIX_RE, "");

    const aux = rest.match(AUX_PREFIX_RE);
    if (aux) {
      sawModal = true;
      rest = rest.slice(aux[0].length);
    } else if (!sawModal) {
      // 「准备/打算」只有出现在助动词之前才算意愿词
      const plan = rest.match(PLAN_PREFIX_RE);
      if (plan) {
        sawModal = true;
        rest = rest.slice(plan[0].length);
      }
    }

    rest = rest.replace(DEADLINE_SUFFIX_RE, "");
    rest = rest.replace(ADV_PREFIX_RE, "");
    if (rest === before) break;
  }

  rest = rest
    .replace(/^[，,、：:\s]+/, "")
    .replace(/[，,、：:。.]+$/, "")
    .replace(/(?:了|啦)+$/, "");

  // 「下午三点和小东开会」这种省了助动词的口语：有时刻、剩的是事情、又不带疑问词，也算一件事。
  // 只说日期（「下周三之前把复盘写完」）同样算，但正文要够长——「明天再说」这类闲聊不算。
  const looksLikeTask =
    !QUESTION_TAIL_RE.test(rest) &&
    ((clock !== "" && rest.length >= 2) || (sawDate && rest.length >= 3));
  if (!sawModal && !looksLikeTask) return null;
  return rest.length >= 2 ? `${clock}${rest}` : null;
}

interface Attributes extends Record<string, unknown> {
  priority?: string;
  estimate?: string;
  dueAt?: string;
  dueTime?: string;
}

// 「今天内完成」「明天之前搞定」是整句的截止说明，不是第二件事。
// 判断标准：这一小句去掉日期、范围词和完成动词后什么都不剩。
const DEADLINE_RANGE_RE = /^(?:之内|以内|内|之前|以前|前)/;
const DEADLINE_VERB_RE = /(?:完成|做完|搞定|弄完|弄好|办完)$/;

function deadlineSegment(segment: string): string | null {
  const due = parseDueDate(segment);
  if (!due) return null;
  const left = segment
    .replace(due.matched, "")
    .replace(/[，,。.；;、\s]/g, "")
    .replace(DEADLINE_RANGE_RE, "")
    .replace(DEADLINE_VERB_RE, "");
  return left === "" ? due.date : null;
}

export function stripDeadlineSegments(text: string): { rest: string; dueAt?: string } {
  const chunks = text.split(/([，,；;。])/);
  let dueAt: string | undefined;
  const kept: string[] = [];
  for (const chunk of chunks) {
    if (/^[，,；;。]$/.test(chunk)) {
      kept.push(chunk);
      continue;
    }
    const due = deadlineSegment(chunk.trim());
    if (due) {
      dueAt = due;
      continue;
    }
    kept.push(chunk);
  }
  return { rest: kept.join("").replace(/[，,；;。]{2,}/g, "，"), dueAt };
}

export function extractAttributes(text: string): { rest: string; attrs: Attributes } {
  let rest = text;
  const attrs: Attributes = {};

  const stripped = stripDeadlineSegments(rest);
  if (stripped.dueAt) attrs.dueAt = stripped.dueAt;
  rest = stripped.rest;

  const priority = rest.match(/(高|中|低)优先级|优先级(高|中|低)|(?:标成|设成|定为)(高|中|低)/);
  if (priority) {
    attrs.priority = PRIORITY[(priority[1] ?? priority[2] ?? "中") as keyof typeof PRIORITY];
    rest = rest.replace(priority[0], "");
  }

  const estimate = rest.match(
    /(大|中|小)工作量|工作量(大|中|小)|(?:标成|设成|定为)(大|中|小)工作量/,
  );
  if (estimate) {
    attrs.estimate = ESTIMATE[(estimate[1] ?? estimate[2] ?? "中") as keyof typeof ESTIMATE];
    rest = rest.replace(estimate[0], "");
  }

  const due = parseDueDate(rest);
  if (due) {
    // 「今天内完成」已经给出过截止日，正文里的日期不再覆盖它
    attrs.dueAt ??= due.date;
    rest = rest.replace(due.matched, "");
  }

  // 时刻单独记一份给提醒用，但不从标题里删——标题留着「下午三点」才看得懂
  const clock = parseDueTime(rest);
  if (clock) attrs.dueTime = clock.time;

  rest = rest.replace(/(?:到期|截止|之前)/g, "");
  return { rest, attrs };
}

// 「把X设为…」这类改属性的说法，前面的 X 必须是短引用。
// 否则「下个月初得把房租交了，别忘了设成高优先级」会被当成改属性，答成「没找到任务」。
function isCleanReference(reference: string): boolean {
  const text = reference.trim();
  return text.length > 0 && text.length <= 12 && !/[，,；;。]/.test(text);
}

function notFound(reference: string, tasks: Task[]): Action[] {
  if (tasks.length === 0) {
    return [{ type: "ask", question: "清单里还没有任务。可以说「帮我记一下买牛奶」先记一条。" }];
  }
  if (reference.trim().length === 0) {
    return [{ type: "ask", question: "要处理哪一条？可以说「删掉买牛奶」这样带上标题。" }];
  }
  return [
    {
      type: "ask",
      question: `没找到「${reference}」这条任务，可以说「看看清单」确认一下当前有哪些任务。`,
    },
  ];
}

type MatchScope = "pending" | "done" | "any";

// 从一句话里剥掉动作词，剩下的就是「说的是哪一条任务」
function referenceOf(message: string): string {
  return cleanTitle(
    message
      .replace(DELETE_RE, "")
      .replace(REOPEN_RE, "")
      .replace(COMPLETE_RE, "")
      .replace(RENAME_RE, ""),
  );
}

// 「都做完了」「全部搞定了」「任务都完成了」：没有点名哪一条，指的是所有待办
function isRangeReference(reference: string): boolean {
  if (reference === "") return false;
  const left = reference
    .replace(/(?:都|全部|所有|这些|那些|任务|待办|事项|事情|东西|事儿)/g, "")
    .trim();
  return left === "";
}

function withTask(
  message: string,
  tasks: Task[],
  build: (id: string) => Action,
  scope: MatchScope = "any",
): Action[] {
  const reference = referenceOf(message);
  // 「删除待办」这类没给标题的说法不猜，先问要动哪一条
  if (GENERIC_REFERENCE_RE.test(reference)) {
    if (tasks.length === 0) return notFound("", tasks);
    const options = tasks
      .slice(0, 5)
      .map((task, index) => `${index + 1}. ${task.title}`)
      .join("；");
    return [
      {
        type: "ask",
        question: `要处理哪一条？当前有：${options}。可以说「删掉 ${tasks[0].title}」。`,
      },
    ];
  }
  let match = findTask(reference, tasks);
  // 「买牛奶这条待办」直接匹配不上时，去掉「这条待办」再试一次
  if (!match.task && !match.ambiguous) {
    const stripped = stripTaskSuffix(reference);
    if (stripped !== reference && stripped.length >= 2) {
      const retry = findTask(stripped, tasks);
      if (retry.task || retry.ambiguous) match = retry;
    }
  }
  // 只说了「这个」「它」这类指代，没有可匹配的标题时，退到按状态找唯一候选
  if (!match.task && !match.ambiguous && normalize(reference).length < 2) {
    const pool =
      scope === "pending"
        ? tasks.filter((task) => !task.done)
        : scope === "done"
          ? tasks.filter((task) => task.done)
          : tasks;
    match =
      pool.length > 1
        ? { candidates: pool, ambiguous: true }
        : { task: pool[0], candidates: pool, ambiguous: false };
  }
  if (match.ambiguous) {
    const options = match.candidates.map((task, index) => `${index + 1}. ${task.title}`).join("；");
    return [{ type: "ask", question: `有几条任务比较像，你说的是哪一个？${options}` }];
  }
  if (!match.task) return notFound(reference, tasks);
  return [build(match.task.id)];
}

export function parseRuleActions(text: string, tasks: Task[]): Action[] | null {
  const message = text.trim();

  // 这三类先判：它们听着像「要做一件事」，其实是对清单或对话的操作
  if (CLEAR_DONE_RE.test(message)) {
    const done = tasks.filter((task) => task.done);
    if (done.length === 0) return [{ type: "ask", question: "现在没有已完成的任务。" }];
    return done.map((task) => ({ type: "delete", id: task.id }));
  }

  if (CLEAR_ALL_RE.test(message)) {
    if (tasks.length === 0) return [{ type: "ask", question: "清单已经是空的。" }];
    return tasks.map((task) => ({ type: "delete", id: task.id }));
  }

  if (CLEAR_CHAT_RE.test(message)) {
    return [
      {
        type: "ask",
        question:
          "聊过的内容在页头「清空对话」里清，点两次确认；任务不会跟着删。要删任务就说「删掉买牛奶」，删已完成的可以说「清空已完成」。",
      },
    ];
  }

  for (const pattern of CREATE_PATTERNS) {
    const matched = message.match(pattern);
    if (matched) {
      const { rest, attrs } = extractAttributes(matched[1]);
      const titles = splitTitles(rest);
      if (titles.some((title) => title.length > 200)) {
        return [{ type: "ask", question: "任务标题请控制在 200 字以内。" }];
      }
      if (titles.length > 0) {
        return titles.map((title) => ({ type: "create", title, fields: attrs }));
      }
    }
  }

  const setPriority = message.match(SET_PRIORITY_RE);
  if (setPriority && isCleanReference(setPriority[1])) {
    const fields = { priority: PRIORITY[setPriority[2] as keyof typeof PRIORITY] };
    return withTask(setPriority[1], tasks, (id) => ({ type: "setAttrs", id, fields }));
  }

  const setEstimate = message.match(SET_ESTIMATE_RE);
  if (setEstimate && isCleanReference(setEstimate[1])) {
    const fields = { estimate: ESTIMATE[setEstimate[2] as keyof typeof ESTIMATE] };
    return withTask(setEstimate[1], tasks, (id) => ({ type: "setAttrs", id, fields }));
  }

  for (const pattern of SET_DUE_RES) {
    const matched = message.match(pattern);
    if (matched) {
      const due = parseDueDate(matched[2] ?? "");
      if (due && isCleanReference(matched[1] ?? "")) {
        return withTask(matched[1] ?? "", tasks, (id) => ({
          type: "setAttrs",
          id,
          fields: { dueAt: due.date },
        }));
      }
    }
  }

  if (DUE_QUERY_RE.test(message)) {
    const due = parseDueDate(message);
    if (due) return [{ type: "due", date: due.date }];
    // 只问了「有哪些到期」没说哪一天时，按今天算，避免退化成普通清单
    if (DUE_QUERY_MARKER.test(message)) return [{ type: "due", date: daysFromToday(0) }];
  }

  const rename = message.match(RENAME_RE);
  if (rename) {
    const title = cleanTitle(rename[1]);
    if (title) return withTask(message, tasks, (id) => ({ type: "rename", id, title }));
  }

  if (DELETE_RE.test(message)) {
    return withTask(message, tasks, (id) => ({ type: "delete", id }));
  }

  if (REOPEN_RE.test(message)) {
    return withTask(message, tasks, (id) => ({ type: "reopen", id }), "done");
  }

  if (COMPLETE_RE.test(message)) {
    // 「都做完了」没点名哪一条：清单里的待办全部标完成
    if (isRangeReference(referenceOf(message))) {
      const pending = tasks.filter((task) => !task.done);
      if (pending.length === 0) return [{ type: "ask", question: "清单里没有待办的任务。" }];
      return pending.map((task) => ({ type: "complete", id: task.id }));
    }
    return withTask(message, tasks, (id) => ({ type: "complete", id }), "pending");
  }

  // 「今天有哪些任务？」：带了日期又问有哪些，问的就是那天要做的事
  if (/(?:哪些|有什么|有啥|多少)/.test(message)) {
    const dated = parseDueDate(message);
    if (dated) return [{ type: "due", date: dated.date }];
  }

  if (LIST_RE.test(message)) {
    return [{ type: "list" }];
  }

  // 「我现在要去取快递」「这周末要给车做保养」这类陈述句不带动作词，
  // 但意图就是新增待办；放在最后判断，避免抢掉前面的查询与属性指令。
  const statement = statementTitle(message);
  // 长复合句（多个小句、多个动作）拆起来歧义大，交给模型判断；
  // 规则只保留 parseLongStatementTitle 作为没有模型时的退路。
  if (statement && !isCompoundStatement(message)) {
    const { rest, attrs } = extractAttributes(statement);
    const due = parseDueDate(rest) ?? parseDueDate(message);
    const titles = splitTitles(rest, false);
    if (titles.length > 0 && titles.every((title) => title.length <= 200)) {
      return titles.map((title) => ({
        type: "create",
        title,
        fields: due ? { ...attrs, dueAt: due.date } : attrs,
      }));
    }
  }

  return null;
}

// 一句话里有多个小句、多件事时，规则不硬拆，交给模型
function isCompoundStatement(message: string): boolean {
  if (message.length <= 16) return false;
  return /[，,；;]/.test(message) || /(?:然后|接着|另外|顺便)/.test(message);
}

// 没有模型时的退路：把长句压成一个标题，也不至于一条都建不出来
export function parseLongStatementTitle(message: string): string | null {
  const statement = statementTitle(message);
  if (!statement) return null;
  const { rest } = extractAttributes(statement);
  const titles = splitTitles(rest, false);
  return titles[0] ?? null;
}
```

```ts
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");
  return `${year}-${month}-${day}`;
}

  const date = new Date();

// 所以这类词只在这里取日期，不进 parseDueDate。
export function statedDay(text: string): string | null {
  const due = parseDueDate(text);
  if (due) return due.date;
  if (/今晚|今早|今夜/.test(text)) return daysFromToday(0);
  if (/明早|明晚|明夜/.test(text)) return daysFromToday(1);
  if (mentionsDayPeriod(text)) return daysFromToday(0);
  if (mentionsInstantDay(text)) return daysFromToday(0);
  return null;
}

export function parseDueDate(text: string): DueMatch | null {
  const relative: Array<[RegExp, number]> = [
    [/大后天/, 3],
    [/后天/, 2],
    [/明天|明日/, 1],
    [/今天|今日/, 0],
  ];
  for (const [pattern, offset] of relative) {
    const matched = text.match(pattern);
    if (matched) return { date: daysFromToday(offset), matched: matched[0] };
  }

  // 「周末」按真实日历落到最近的周六；已经到周末就算当天
  const weekend = text.match(/(下下|下|这|本)?周末/);
  if (weekend) {
    const weeks = weekend[1] === "下下" ? 2 : weekend[1] === "下" ? 1 : 0;
    const today = new Date().getDay();
    const delta = (today === 6 || today === 0 ? 0 : 6 - today) + weeks * 7;
    return { date: daysFromToday(delta), matched: weekend[0] };
  }

  const weekday = text.match(/(下下|下|这|本)?(?:周|星期)([一二三四五六日天])/);
  if (weekday) {
    const target = "一二三四五六日天".indexOf(weekday[2] ?? "");
    const today = new Date().getDay();
    const current = today === 0 ? 6 : today - 1;
    const weeks = weekday[1] === "下下" ? 2 : weekday[1] === "下" ? 1 : 0;
    let delta: number;
    if (weeks > 0) {
      // 「下周X」「下下周X」都从本周周一起算，避免在已跨周的结果上再加七天
      delta = 7 * weeks - current + target;
    } else {
      delta = (target - current + 7) % 7;
      // 「这周五」就是本周那天，当天也算；只说「周五」才顺延到下一个
      const thisWeek = weekday[1] === "这" || weekday[1] === "本";
      if (delta === 0 && !thisWeek) delta = 7;
    }
    return { date: daysFromToday(delta), matched: weekday[0] };
  }

  // 「月底/下个月初」这类说法按真实日历落到当月最后一天或下月一号。
  // 必须带「月」字：「彻底打扫」里的「底」不是月底。
  const monthEdge = text.match(/下个?月(底|末|初)/) ?? text.match(/(?:本|这个)?月(底|末|初)/);
  if (monthEdge) {
    const now = new Date();
    const isNext = text.includes("下个月") || text.includes("下月");
    const offset = isNext ? 1 : 0;
    const unit = monthEdge[1];
    const first = new Date(now.getFullYear(), now.getMonth() + offset, 1);
    let date = unit === "初" ? first : new Date(now.getFullYear(), now.getMonth() + offset + 1, 0);
    // 「月初」已经过去就顺延到下个月一号，不记一个过期的时间
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    if (unit === "初" && date < today) {
      date = new Date(now.getFullYear(), now.getMonth() + 1, 1);
    }
    return { date: localDate(date), matched: monthEdge[0] };
  }

  const monthDay = text.match(/(\d{1,2})月(\d{1,2})[日号]?/);
  if (monthDay) {
    const month = Number(monthDay[1]);
    const day = Number(monthDay[2]);
    const now = new Date();
    let year = now.getFullYear();
    const candidate = new Date(year, month - 1, day);
    if (candidate < new Date(now.getFullYear(), now.getMonth(), now.getDate())) year += 1;
    return { date: localDate(new Date(year, month - 1, day)), matched: monthDay[0] };
  }

  return null;
}

// 日期只写数字，跨年才带年份
export function shortDate(date: string): string {
  const matched = date.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!matched) return date;
  return Number(matched[1]) === new Date().getFullYear() ? `${matched[2]}-${matched[3]}` : date;
}

// 口头提一句日期时用的说法
export function dayLabel(date: string): string {
  if (date === daysFromToday(0)) return "今天";
  if (date === daysFromToday(1)) return "明天";
  if (date === daysFromToday(2)) return "后天";
  return shortDate(date);
}

// 期限统一说成「11-28 内完成」：日期在前，不同日期的待办一眼分得清
export function dueBadge(date: string): string {
  return `${shortDate(date)} 内完成`;
}

const CN_DIGIT: Record<string, number> = {
  一: 1,
  二: 2,
  三: 3,
  四: 4,
  五: 5,
  六: 6,
  七: 7,
  八: 8,
  九: 9,
  十: 10,
};
// 时段词把 12 小时制折算成 24 小时制；凌晨/早上/上午保持原值
const NOON_PERIODS = new Set(["中午", "下午", "傍晚", "晚上"]);

function toHour(raw: string): number | null {
  if (/^\d{1,2}$/.test(raw)) return Number(raw);
  if (raw === "十一") return 11;
  if (raw === "十二") return 12;
  return CN_DIGIT[raw] ?? null;
}

// 从原话里单独取出时刻，写进 dueTime，用来做提醒。
// 标题里仍然保留「下午三点」，这里只是额外记一份结构化的时间。
export function parseDueTime(text: string): { time: string; matched: string } | null {
  const matched = text.match(
    /(上午|早上|凌晨|中午|下午|傍晚|晚上)?\s*(\d{1,2}|[一二三四五六七八九十]{1,2})\s*[点时](半|\d{1,2}分)?/,
  );
  if (!matched?.[2]) return null;

  const period = matched[1];
  let hour = toHour(matched[2]);
  if (hour === null) return null;

  if (period && NOON_PERIODS.has(period) && hour < 12) hour += 12;
  // 「三点开会」这种没有时段词的说法，口语里通常指下午，1-6 点按下午折算
  if (!period && hour >= 1 && hour <= 6) hour += 12;
  if (hour > 23) return null;

  const suffix = matched[3];
  const minute = suffix === "半" ? 30 : suffix ? Number(suffix.replace("分", "")) : 0;
  if (!Number.isFinite(minute) || minute > 59) return null;

  return {
    time: `${String(hour).padStart(2, "0")}:${String(minute).padStart(2, "0")}`,
    matched: matched[0].trim(),
  };
}
```

```ts
const TIME_WORD_RE =

    .replace(/[，,。.！!？?；;、\s]/g, "")

const ANSWER_FILLER_RE = /^(?:那|就|也是|是|都|肯定|当然|其实|反正|要|得|应该)+/;

function todayAnswer(message: string): boolean {
  let core = message.replace(/现在|马上|立刻|等下|待会|待会儿|一会儿|晚点|稍后/g, "").trim();
  for (;;) {
    const before = core;
    core = core.replace(ANSWER_FILLER_RE, "");
    core = core.replace(/[啊呀吧呢哦哈呗了啦嘛]+$/, "");
    core = core.trim();
    if (core === before) break;
  }
  return /^(?:今天|今日|当天|今儿)(?:晚上|早上|上午|下午|中午|夜里|凌晨)?(?:之内|以内|内)?(?:完成|做完|搞定|办完|结束)?$/.test(

}

    if (timeOnlyRest(message) !== "") return { kind: "other", text: message };

  if (clock && timeOnlyRest(message) === "") return { kind: "today", time: clock.time };
  return { kind: "other", text: message };
```

```ts
import type { Task } from "../task-store.ts";

const PRIORITY_LABEL = { high: "高优先级", medium: "中优先级", low: "低优先级" } as const;
const ESTIMATE_LABEL = { large: "大工作量", medium: "中工作量", small: "小工作量" } as const;

function tagText(task: Task): string {
  const tags: string[] = [];
  const priority = task.priority;
  const estimate = task.estimate;
  if (typeof priority === "string") {
    tags.push(PRIORITY_LABEL[priority as keyof typeof PRIORITY_LABEL] ?? priority);
  }
  if (typeof estimate === "string") {
    tags.push(ESTIMATE_LABEL[estimate as keyof typeof ESTIMATE_LABEL] ?? estimate);
  }
  if (typeof task.detail === "string") tags.push(task.detail);
  return tags.length > 0 ? ` [${tags.join(" · ")}]` : "";
}

function rowText(task: Task, index: number): string {
  return `${task.done ? "☑" : "☐"} ${index + 1}. ${task.title}${tagText(task)}`;
}

function renderRows(tasks: Task[]): string {
  return tasks.map((task, index) => rowText(task, index)).join("\n");
}

interface DayGroup {
  date: string;
  label: string;
  tasks: Task[];
}

// 同一天的任务并到一组，日期只写一次，不逐条重复
function groupByDay(tasks: Task[]): DayGroup[] {
  const dated = new Map<string, Task[]>();
  const undated: Task[] = [];
  for (const task of tasks) {
    if (typeof task.dueAt !== "string") {
      undated.push(task);
      continue;
    }
    const bucket = dated.get(task.dueAt) ?? [];
    bucket.push(task);
    dated.set(task.dueAt, bucket);
  }

  const groups = [...dated.entries()]
    .sort(([a], [b]) => a.localeCompare(b))
    .map(([date, items]) => ({ date, label: dueBadge(date), tasks: items }));
  if (undated.length > 0) groups.push({ date: "", label: "未排期", tasks: undated });
  return groups;
}

export function renderList(tasks: Task[]): string {
  if (tasks.length === 0) return "清单是空的，还没有待办。";
  const pending = tasks.filter((task) => !task.done).length;
  // 一条都没剩下时不说「其中 0 项待办」，读着别扭
  const header =
    pending === 0
      ? `共 ${tasks.length} 项，都已完成：`
      : `共 ${tasks.length} 项，其中 ${pending} 项待办：`;
  const groups = groupByDay(tasks);

  // 一条都没排期时不用多一个「未排期」标题
  const only = groups[0];
  if (groups.length === 1 && only && only.date === "") {
    return [header, renderRows(only.tasks)].join("\n");
  }

  let seen = 0;
  const blocks = groups.map((group) => {
    const rows = group.tasks
      .map((task) => {
        seen += 1;
        return rowText(task, seen - 1);
      })
      .join("\n");
    return `${group.label}（${group.tasks.length} 项）\n${rows}`;
  });
  return [header, ...blocks].join("\n\n");
}

export function formatResults(results: ActionResult[], tasks: Task[]): string {
  for (const result of results) {
    if (result.action.type === "ask") return result.action.question;
  }

  const dueResult = results.find((result) => result.action.type === "due");
  if (dueResult?.action.type === "due") {
    const dueTasks = dueResult.tasks ?? [];
    const day = shortDate(dueResult.action.date);
    if (dueTasks.length === 0) return `${day} 没有需要完成的待办。`;
    return [`${day} 有 ${dueTasks.length} 项待办需要完成：`, renderRows(dueTasks)].join("\n");
  }

  const parts: string[] = [];
  for (const { action, task } of results) {
    switch (action.type) {
      case "create":
        if (task) parts.push(`已记下「${task.title}」。`);
        break;
      case "complete":
        parts.push(task ? `已把「${task.title}」标记为完成。` : "没找到这条任务。");
        break;
      case "reopen":
        parts.push(task ? `已把「${task.title}」重新标为待办。` : "没找到这条任务。");
        break;
      case "rename":
        parts.push(task ? `已改名为「${task.title}」。` : "没找到这条任务。");
        break;
      case "setAttrs":
        parts.push(task ? `已更新「${task.title}」的属性。` : "没找到这条任务。");
        break;
      case "delete":
        parts.push(task ? `已删除「${task.title}」。` : "没找到这条任务。");
        break;
      case "list":
      case "due":
      case "ask":
        break;
    }
  }

  const list = renderList(tasks);
  return parts.length > 0 ? `${parts.join("\n")}\n\n${list}` : list;
}
```

```ts
fields 可含 priority(high|medium|low)、estimate(small|medium|large)、dueAt(YYYY-MM-DD)。
只填用户明确说过或能从原话直接得到的字段：没提优先级就不要给 priority，没提工作量就不要给 estimate，
没说哪天完成就不要给 dueAt（留空由程序按当天处理），不要自己推断轻重缓急。
规则：
- id 必须来自下面清单，禁止编造。
- 用户用「牛奶买好了」这类省略说法时，要根据标题模糊匹配到对应任务。
- 匹配不唯一或找不到时，用 ask 反问用户，不要猜。
- 一句话里有几个独立动作就输出几条 create：用「然后 / 接着 / 再 / 另外 / 顺便」连起来的动作各算一条，不要合成一条。
  例：「下周三之前把复盘写完，然后发给张老师」→ 两条：写完复盘、把复盘发给张老师。
- 拆出来的每条标题要能独立看懂，省略的宾语按上文补全（上例第二条补成「把复盘发给张老师」），但不要加用户没说的内容。
- 标题要保留用户说的时间点（上午十点、下午三点），只把日期放进 dueAt，不要连时刻一起删掉。
- 只有一个动作时不要硬拆。
- 今天/明天等相对日期按下面给出的今天日期换算成 YYYY-MM-DD。
- 日期拿不准时也要先用 create 把任务记下来（dueAt 留空），再用 ask 追问日期；
  不要因为「哪天做」不确定就把任务丢掉。
- 日历上的安排和用户说法冲突时（比如用户说下周一，但日历显示是周二），create 仍按用户说的记，再用 ask 指出冲突。
- 跟用户提到期限时说「11-28 内完成」这种写法（数字日期 + 内完成），不要说「到期」。
- 不执行任何与待办无关的请求，用 ask 说明能力范围。`;

// 系统提示词里必须带上当天日期，否则模型无法把「明天」换算成具体日期。
// 有日历时一并给出来，「周末」「下周」这类不具体的说法才能落到真实可用的日子。
export function systemPrompt(now = new Date(), calendar = calendarContext()): string {
  const calendarBlock = calendar
    ? `\n${calendar}\n用户说「周末」「下周」「月底」这类不具体的时间时，按上面的日历挑一天；如果候选的日子都已经排满，用 ask 反问用户选哪天。\n`
    : "";
  return `${PROMPT_BODY}\n今天是 ${localDate(now)}（${WEEKDAYS[now.getDay()]}）。${calendarBlock}\n当前清单：\n`;
}

// 模型偶尔把 type 写成 action，这里统一成 type 再交给 schema 校验。
function normalizeAction(raw: unknown): unknown {
  if (!raw || typeof raw !== "object") return raw;
  const copied = { ...(raw as Record<string, unknown>) };
  if (typeof copied.type !== "string" && typeof copied.action === "string") {
    copied.type = copied.action;
    delete copied.action;
  }
  return copied;
}

// 把模型返回的文本解析成动作列表；JSON 不合规、结构不符时返回 null，由调用方回退规则引擎。
export function parsePlan(content: string): Action[] | null {
  try {
    const data: unknown = JSON.parse(content);
    const list = Array.isArray(data)
      ? data
      : ((data as { actions?: unknown } | null)?.actions as unknown);
    if (!Array.isArray(list)) return null;
    const parsed = planSchema.safeParse({ actions: list.map(normalizeAction) });
    return parsed.success ? parsed.data.actions : null;
  } catch {
    return null;
  }
}

export async function planWithLLM(message: string, tasks: Task[]): Promise<Action[] | null> {
  if (!process.env.DEEPSEEK_API_KEY) return null;
  try {
    const client = getModelClient();
    const snapshot =
      tasks
        .map((task) => `- ${task.id} [${task.done ? "已完成" : "待办"}] ${task.title}`)
        .join("\n") || "（空）";
    const completion = await client.chat.completions.create({
      model,
      temperature: 0,
      response_format: { type: "json_object" },
      messages: [
        { role: "system", content: systemPrompt() + snapshot },
        { role: "user", content: message },
      ],
    });
    const content = completion.choices[0]?.message?.content;
    if (!content) return null;
    return parsePlan(content);
  } catch (error) {
    console.error("模型规划失败，已回退到本地规则引擎：", error);
    return null;
  }
}
```

### `server/agent/calendar.ts`

iCalendar 日历解析，用于撞期提醒（106 行）

```ts
import { existsSync, readFileSync, statSync } from "node:fs";
import { localDate } from "./date.ts";

export interface CalendarEvent {
  date: string;
  title: string;
  allDay: boolean;
}

// 日历来源：优先环境变量指定的文件，其次 data/calendar.ics。
// 用 .ics 而不是各家日历 API，是为了不依赖授权——用户从手机或电脑导出即可。
// 没有这个文件就当作没有日历，不做任何提醒；格式示例见 examples/calendar.example.ics。
const DEFAULT_PATHS = ["data/calendar.ics", "data/calendar.ical"];

interface Cache {
  key: string;
  events: CalendarEvent[];
}
let cache: Cache | null = null;

function unescapeText(value: string): string {
  return value
    .replace(/\\n/gi, " ")
    .replace(/\\,/g, ",")
    .replace(/\\;/g, ";")
    .replace(/\\\\/g, "\\");
}

function toDate(value: string): string {
  return `${value.slice(0, 4)}-${value.slice(4, 6)}-${value.slice(6, 8)}`;
}

// 只取 VEVENT 的开始日期与标题；够用来判断「那天有没有安排」。
export function parseIcs(text: string): CalendarEvent[] {
  const unfolded = text.replace(/\r?\n[ \t]/g, "");
  const events: CalendarEvent[] = [];

  for (const chunk of unfolded.split(/BEGIN:VEVENT/).slice(1)) {
    const body = chunk.split(/END:VEVENT/)[0] ?? "";
    const start = body.match(/DTSTART[^:\r\n]*:(\d{8})/);
    if (!start?.[1]) continue;
    const summary = body.match(/SUMMARY:(.+)/);
    events.push({
      date: toDate(start[1]),
      title: summary?.[1] ? unescapeText(summary[1].trim()) : "（未命名日程）",
      allDay: /DTSTART;VALUE=DATE/.test(body),
    });
  }

  return events.sort((a, b) => a.date.localeCompare(b.date));
}

export function calendarPath(): string | null {
  const fromEnv = process.env.CALENDAR_ICS_PATH;
  if (fromEnv && existsSync(fromEnv)) return fromEnv;
  return DEFAULT_PATHS.find((path) => existsSync(path)) ?? null;
}

// 按文件修改时间缓存，改了日历不用重启
export function loadCalendar(): CalendarEvent[] {
  const path = calendarPath();
  if (!path) return [];

  const key = `${path}:${statSync(path).mtimeMs}`;
  if (cache?.key === key) return cache.events;

  try {
    const events = parseIcs(readFileSync(path, "utf8"));
    cache = { key, events };
    return events;
  } catch (error) {
    console.error("日历读取失败，按没有日历处理：", error);
    return [];
  }
}

export function eventsOn(date: string): CalendarEvent[] {
  return loadCalendar().filter((event) => event.date === date);
}

// 把近期安排压成几行，交给模型判断「周末/下周」落到哪天更合适。
export function calendarContext(days = 14): string {
  const events = loadCalendar();
  if (events.length === 0) return "";

  const today = localDate(new Date());
  const limit = new Date();
  limit.setDate(limit.getDate() + days);
  const until = localDate(limit);

  const upcoming = events.filter((event) => event.date >= today && event.date <= until);
  if (upcoming.length === 0) return "";

  const lines = upcoming.map(
    (event) => `- ${event.date} ${event.title}${event.allDay ? "（全天）" : ""}`,
  );
  return `日历里近期已有安排（判断日期时要避开已占满的日子）：\n${lines.join("\n")}`;
}

export function calendarNote(date: string): string {
  const events = eventsOn(date);
  if (events.length === 0) return "";
  const titles = events.map((event) => event.title).join("、");
  return `\n\n（提醒：${date} 日历上已有 ${events.length} 项安排：${titles}）`;
}
```

```ts
import { resolve } from "node:path";
import express, { type ErrorRequestHandler, type Request } from "express";
import { chatRequestSchema, type ChatMessage } from "../shared/chat.ts";
import type { TaskStore } from "./task-store.ts";
import { taskRouter } from "./tasks.ts";
import { buildReminders } from "./reminders.ts";
import { normalizeSessionId, type ConversationStore } from "./conversations.ts";
import { normalizeUserId } from "./users.ts";

interface AppDependencies {
  reply(messages: ChatMessage[], tasks: TaskStore): Promise<string>;
  tasks: TaskStore;
  /** 按匿名用户 id 取各自的清单；不传时所有人都用同一个 tasks（单用户模式） */
  storeFor?(userId: string): TaskStore;
  /** 对话落盘；不传时对话只活在请求里，不持久化 */
  conversations?: ConversationStore;
}

export function createApp({ reply, tasks, storeFor, conversations }: AppDependencies) {
  const app = express();
  app.use(express.json({ limit: "1mb" }));

  const storeOf = (request: Request): TaskStore =>
    storeFor ? storeFor(normalizeUserId(request.get("x-user-id"))) : tasks;

  app.post("/api/chat", async (request, response) => {
    const input = chatRequestSchema.safeParse(request.body);
    if (!input.success) {
      return response.status(400).json({ error: "请提供对话记录，并以用户消息结尾。" });
    }

    const userId = normalizeUserId(request.get("x-user-id"));
    const sessionId = normalizeSessionId(request.get("x-session-id"));
    const messages = input.data.messages;

    // 先把用户这句落盘，模型挂了也不丢
    if (sessionId && conversations) conversations.save(userId, sessionId, messages);

    const answer = await reply(messages, storeFor ? storeFor(userId) : tasks);

    if (sessionId && conversations) {
      conversations.save(userId, sessionId, [...messages, { role: "assistant", content: answer }]);
    }
    response.json({ reply: answer });
  });

  // 刷新页面后从这里把最近若干轮捞回来
  app.get("/api/conversations/:sessionId", (request, response) => {
    const sessionId = normalizeSessionId(request.params.sessionId);
    if (!sessionId || !conversations) return response.json({ messages: [] });

    const userId = normalizeUserId(request.get("x-user-id"));
    response.json({ messages: conversations.recent(userId, sessionId) });
  });

  app.delete("/api/conversations/:sessionId", (request, response) => {
    const sessionId = normalizeSessionId(request.params.sessionId);
    if (sessionId && conversations) {
      conversations.clear(normalizeUserId(request.get("x-user-id")), sessionId);
    }
    response.sendStatus(204);
  });

  // 提醒条和浏览器通知的数据源：逾期、今天、一小时内到点
  app.get("/api/reminders", (request, response) => {
    response.json(buildReminders(storeOf(request).list()));
  });

  app.use(
    "/api/tasks",
    taskRouter((request) => storeOf(request)),
  );
  app.use("/api", (_req, res) => res.status(404).json({ error: "接口不存在。" }));
  app.use(handleApiError);

  app.use(express.static(resolve("public")));
  app.get("/assets/chat.js", (_req, res) => res.sendFile(resolve("dist/chat.js")));
  return app;
}

const handleApiError: ErrorRequestHandler = (error, _req, res, _next) => {
  if (error.type === "entity.parse.failed") {
    return res.status(400).json({ error: "请求需要使用有效的 JSON。" });
  }
  if (error.type === "entity.too.large") {
    return res.status(413).json({ error: "对话太长，请刷新后开始新的对话。" });
  }
  console.error(error);
  res.status(500).json({ error: "服务暂时无法处理请求，请稍后重试。" });
};
```

### `server/task-store.ts`

本地 JSON 任务清单：looseObject 透传扩展字段（61 行）

```ts
import { randomUUID } from "node:crypto";
import { existsSync, mkdirSync, readFileSync, renameSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";
import { z } from "zod";

const storedTask = z.looseObject({ id: z.string(), title: z.string(), done: z.boolean() });
const taskList = z.array(storedTask);
export type Task = z.infer<typeof storedTask>;
export type TaskChanges = Partial<Pick<Task, "title" | "done">> & Record<string, unknown>;

export interface TaskStore {
  list(): Task[];
  create(title: string, extra?: Record<string, unknown>): Task;
  update(id: string, changes: TaskChanges): Task | undefined;
  remove(id: string): boolean;
}

export function createJsonTaskStore(filename: string): TaskStore {
  function list(): Task[] {
    if (!existsSync(filename)) return [];
    return taskList.parse(JSON.parse(readFileSync(filename, "utf8")));
  }

  // 同一服务进程内同步完成读写；替换文件前，原清单始终完整可读。
  function save(tasks: Task[]) {
    mkdirSync(dirname(filename), { recursive: true });
    const temporary = filename + ".tmp";
    writeFileSync(temporary, JSON.stringify(tasks, null, 2) + "\n");
    renameSync(temporary, filename);
  }

  return {
    list,

    create(title, extra = {}) {
      const task = { ...extra, id: randomUUID(), title, done: false };
      save([...list(), task]);
      return task;
    },

    update(id, changes) {
      const tasks = list();
      const task = tasks.find((item) => item.id === id);
      if (!task) return undefined;

      Object.assign(task, changes);
      save(tasks);
      return task;
    },

    remove(id) {
      const tasks = list();
      const remaining = tasks.filter((task) => task.id !== id);
      if (remaining.length === tasks.length) return false;

      save(remaining);
      return true;
    },
  };
}
```
